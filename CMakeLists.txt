cmake_minimum_required(VERSION 3.20)
project(PixelShield LANGUAGES CXX)
add_executable(PixelShield WIN32 main.cpp)
target_compile_features(PixelShield PRIVATE cxx_std_17)
target_compile_definitions(PixelShield PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(PixelShield PRIVATE user32 gdi32 shell32 comctl32 ole32 uuid dxva2 advapi32)
if(MSVC)
  target_compile_options(PixelShield PRIVATE /O1 /W4 /EHsc)
else()
  target_compile_options(PixelShield PRIVATE -Os -Wall)
  target_link_options(PixelShield PRIVATE -municode -mwindows)
endif()
